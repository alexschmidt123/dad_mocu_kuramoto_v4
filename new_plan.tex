\section*{System Model and Latent Uncertainty (IEEE-14)}

\textbf{Network-coupled swing equation (second-order Kuramoto):}
\[
\dot{\theta}_i(t) = \omega_i(t)
\]

\[
M\,\dot{\omega}_i(t)
=
P_{m,i}
-
\sum_{j=1}^{N} B_{ij}\sin\!\big(\theta_i(t)-\theta_j(t)\big)
-
D\,\omega_i(t)
-
K\,\omega_i(t)
+
u^{\text{probe}}_{\xi,i}(t)
+
u^{\text{ctrl}}_{\gamma,i}(t)
\]

\vspace{0.3em}
\textbf{Pairwise electrical coupling (explicit):}
\[
P_{e,i}(\theta)
=
\sum_{j=1}^{N} B_{ij}\sin(\theta_i-\theta_j)
\quad\Rightarrow\quad
\text{direct pair interactions }(i,j)
\]

\vspace{0.3em}
\textbf{Latent parameters (uncertain, fixed during operation):}
\[
\vartheta = (M, K)\in\mathbb{R}_+^2
\]

\begin{itemize}
  \item $M$: equivalent inertia (aggregate inertia of the system)
  \item $K$: equivalent fast frequency response / primary control gain
\end{itemize}

\textbf{Known / fixed:}
\begin{itemize}
  \item Network coupling $B_{ij}$ (IEEE-14)
  \item Damping $D$, nominal operating point $P_{m,i}$
\end{itemize}

\vspace{0.3em}
\textbf{Key modeling point:}

Uncertainty in $(M,K)$ affects how much frequency control is required,
not the form of the dynamics.

\section*{Control Decision $\gamma$ and Frequency Security}

\textbf{Primary / fast frequency control model:}
\[
u^{\text{ctrl}}_{\gamma,i}(t)
=
\gamma\, g_i\,\omega_i(t),
\qquad
\sum_i g_i = 1,\ g_i\ge 0
\]

\begin{itemize}
  \item $\gamma$: total committed primary / fast frequency control capacity
  \item $g_i$: spatial allocation of control across buses
\end{itemize}

\vspace{0.3em}
\textbf{Interpretation:}

$\gamma$ is a \emph{planning-level decision} chosen before disturbances,
not a state or a parameter to be estimated.

\vspace{0.4em}
\textbf{Critical control requirement for fixed $(M,K)$:}
\[
\gamma^*(M,K)
=
\min_{\gamma}
\quad \text{s.t. frequency constraints are satisfied}
\]

Example constraints:
\[
\max_t |\dot{f}(t)| \le r_{\max},
\qquad
\min_t f(t) \ge f_{\min}
\]

\textbf{Computation:}

$\gamma^*(M,K)$ is obtained via simulation of the swing equation
and binary search over $\gamma$.

\vspace{0.3em}
\textbf{Monotonicity:}

Lower $M$ or lower $K$ $\Rightarrow$ larger $\gamma^*(M,K)$.

\section*{Experiment, Observation, and Belief Update}

\textbf{Experiment (active probing action):}
\[
\xi = (b,A,T)
\]

\[
u^{\text{probe}}_{\xi,i}(t)=
\begin{cases}
A\,s(t;T), & i=b,\\
0, & i\neq b,
\end{cases}
\]

with Hann window:
\[
s(t;T)=
\frac{1}{2}\left(1-\cos\frac{2\pi t}{T}\right),\quad 0\le t\le T
\]

\begin{itemize}
  \item $b$: bus location ($1,\dots,14$)
  \item $A$: probe amplitude
  \item $T=2$ s (literature-aligned)
\end{itemize}

\vspace{0.3em}
\textbf{Observation (PMU-like):}
\[
\Delta f_i(t)=\frac{\omega_i(t)}{2\pi},
\qquad f_s=12~\text{Hz},\quad t\in[0,10]\text{ s}
\]

\textbf{Feature extraction:}
\[
y = \phi(\Delta f(t))
=
[\text{ROCOF}_{\max},\ f_{\min},\ t_{\text{settle}},\dots]
\]

\vspace{0.3em}
\textbf{Belief update:}
\[
p_{t+1}(M,K)
\propto
p(y_t\mid M,K,\xi_t)\,p_t(M,K)
\]

Estimation is \emph{intermediate}, not the final objective.

\section*{MOCU and Sequential Experimental Design}

\textbf{Robust control under uncertainty:}
\[
\mathcal{A}_t=\text{support / credible set of }p_t(M,K)
\]

\[
\gamma^*(\mathcal{A}_t)
=
\max_{(M,K)\in\mathcal{A}_t}\gamma^*(M,K)
\]

\vspace{0.3em}
\textbf{Mean Objective Cost of Uncertainty (MOCU):}
\[
\mathrm{MOCU}(p_t)
=
\mathbb{E}_{(M,K)\sim p_t}
\Big[
\gamma^*(\mathcal{A}_t)
-
\gamma^*(M,K)
\Big]
\]

\textbf{Interpretation:}

MOCU measures the \emph{expected excess primary control}
required due to uncertainty in $(M,K)$.

\vspace{0.4em}
\textbf{Sequential design principle:}
\[
\xi_{t+1}
=
\arg\min_{\xi\in\Xi}
\mathbb{E}\big[\mathrm{MOCU}(p_{t+1})\big]
\]

\textbf{Efficiency motivation:}

Myopic optimization over large $\Xi$ is slow;
we learn an amortized DAD/iDAD policy
to map belief $\rightarrow$ probe directly.

\vspace{0.4em}
\textbf{Practical data-generation plan (IEEE-14):}
\begin{itemize}
  \item Sample $(M,K)$ from continuous prior ranges
  \item Sample probes $\xi=(b,A,T)$ from discrete design set
  \item Simulate swing equation (GPU-batched)
  \item Extract frequency features $y$
  \item Compute $\gamma^*(M,K)$ by binary search
  \item Store $(M,K,\xi,y,\gamma^*)$ for MOCU and policy training
\end{itemize}
